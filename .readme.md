# Clone your fork
git clone https://github.com/vikasuxa/MeshCentral.git
cd MeshCentral

# Track original repo
git remote add upstream https://github.com/Ylianst/MeshCentral.git

# Create custom branch
git checkout -b customize

# Make changes, then:
git add .
git commit -m "My changes"
git push origin customize

# To update later:
git checkout master
git pull upstream master
git checkout customize
git merge master
